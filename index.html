// WMS Pro - Database Google Sheet (VERSI 2: simpan per baris, aman untuk banyak pengguna)
function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = {};
  ss.getSheets().forEach(function (sh) {
    data[sh.getName()] = sh.getLastRow() === 0 ? [] : sh.getDataRange().getValues();
  });
  return json_({ status: 'connected', spreadsheet: ss.getName(), sheets: Object.keys(data), data: data });
}

function ensureSheet_(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  if (headers.length > 0) {
    if (headers.length > sheet.getMaxColumns()) {
      sheet.insertColumnsAfter(sheet.getMaxColumns(), headers.length - sheet.getMaxColumns());
    }
    var cur = sheet.getRange(1, 1, 1, headers.length).getValues()[0];
    if (cur.join('|') !== headers.join('|')) {
      sheet.getRange(1, 1, 1, headers.length).setNumberFormat('@').setValues([headers]).setFontWeight('bold');
      sheet.setFrozenRows(1);
    }
  }
  return sheet;
}

// Kolom berisi teks (tanggal, jam, kode) disimpan sebagai TEXT agar tidak diubah otomatis oleh Sheets
function writeBlock_(sheet, startRow, rows) {
  if (rows.length === 0) return;
  var cols = 0;
  rows.forEach(function (r) { if (r.length > cols) cols = r.length; });
  var norm = rows.map(function (r) {
    var a = r.slice();
    while (cols - a.length > 0) a.push('');
    return a;
  });
  if (cols > sheet.getMaxColumns()) sheet.insertColumnsAfter(sheet.getMaxColumns(), cols - sheet.getMaxColumns());
  var needRows = startRow + norm.length - 1;
  if (needRows > sheet.getMaxRows()) sheet.insertRowsAfter(sheet.getMaxRows(), needRows - sheet.getMaxRows());
  for (var c = 0; c !== cols; c++) {
    var hasText = norm.some(function (r) { return typeof r[c] !== 'number'; });
    if (hasText) sheet.getRange(startRow, c + 1, norm.length, 1).setNumberFormat('@');
  }
  sheet.getRange(startRow, 1, norm.length, cols).setValues(norm);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(25000);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    var p = JSON.parse(raw);
    var name = p.sheetName || 'Log_Aktivitas';
    var action = p.action || 'upsert';
    var headers = p.headers || [];
    var rows = p.rows || [];
    var sheet = ensureSheet_(ss, name, headers);
    var changed = 0;
    var last = sheet.getLastRow();

    if (action === 'replace_all') {
      if (last > 1) sheet.getRange(2, 1, last - 1, sheet.getMaxColumns()).clearContent();
      writeBlock_(sheet, 2, rows);
      changed = rows.length;
    } else if (action === 'delete') {
      if (last > 1) {
        var ids = sheet.getRange(2, 1, last - 1, 1).getValues();
        for (var i = ids.length - 1; i >= 0; i--) {
          if (String(ids[i][0]) === String(p.id)) { sheet.deleteRow(i + 2); changed++; }
        }
      }
    } else if (action === 'insert') {
      writeBlock_(sheet, Math.max(last, 1) + 1, rows);
      changed = rows.length;
    } else {
      // upsert: ID sama = perbarui baris, ID baru = tambah di bawah
      var existing = {};
      if (last > 1) {
        sheet.getRange(2, 1, last - 1, 1).getValues().forEach(function (r, idx) {
          existing[String(r[0])] = idx + 2;
        });
      }
      var appendMap = {};
      var appendList = [];
      rows.forEach(function (r) {
        var key = String(r[0]);
        if (existing[key]) {
          writeBlock_(sheet, existing[key], [r]);
        } else if (appendMap[key] !== undefined) {
          appendList[appendMap[key]] = r;
        } else {
          appendMap[key] = appendList.length;
          appendList.push(r);
        }
      });
      writeBlock_(sheet, Math.max(sheet.getLastRow(), 1) + 1, appendList);
      changed = rows.length;
    }
    return json_({ status: 'ok', sheetName: name, action: action, changed: changed });
  } catch (err) {
    return json_({ status: 'error', message: err.toString() });
  } finally {
    try { lock.releaseLock(); } catch (x) {}
  }
}
